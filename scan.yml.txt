name: فحص العملات الحلال

on:
  schedule:
    # يعمل كل ساعة (بتوقيت UTC)
    - cron: "5 * * * *"
  workflow_dispatch: {}   # يسمح بتشغيل يدوي من تبويب Actions

jobs:
  scan:
    runs-on: ubuntu-latest
    steps:
      - name: سحب الكود
        uses: actions/checkout@v4

      - name: تثبيت Python
        uses: actions/setup-python@v5
        with:
          python-version: "3.11"

      - name: تثبيت المكتبات
        run: pip install pandas requests

      - name: تشغيل الماسح
        env:
          TELEGRAM_BOT_TOKEN: ${{ secrets.TELEGRAM_BOT_TOKEN }}
          TELEGRAM_CHAT_ID: ${{ secrets.TELEGRAM_CHAT_ID }}
        run: python scanner.py
